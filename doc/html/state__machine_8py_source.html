231 def _create_state_list(self, data:dict) -> list[State]:
232 result: list[State] = []
233 for state_data in data:
234 # create State
235 tmp = State(state_data["name"], state_data["parent"])
236 # add state hierarchy
237 default_child_idx = state_data["default_child"]
238 for idx, child_name in enumerate(state_data["children"]):
239 if idx == state_data["default_child"]:
240 tmp.add_child(child_name, True)
241 else:
242 tmp.add_child(child_name)
243 # add transition info
244 for trans_data in state_data["transitions"]:
245 trans = TransitionInfo(trans_data["target"], trans_data["trigger"])
246 tmp.add_transition(trans)
247
248 # define activity
249 tmp.define_entry_activity(state_data["do_entry"])
250 tmp.define_while_activity(state_data["do_while"])
251 tmp.define_exit_activity(state_data["do_exit"])
252
253 result.append(tmp)
254 return result

256 def _check_transition(self, event: str) -> int:
257 for idx in self._current_active_state_idx_list:
258 state = self._state_list[idx]
259 for transition in state._transitions:
260 if transition._trigger == event and transition.condition():
261 idx = self._search_state_idx(transition._target)
262 if idx >= 0:
263 return idx
264 return -1

266 def _get_next_active_state_list(self, pivot_state: str) -> list[int]:
267 tmp_list = []
268 # add pivot state
269 pivot_idx = self._search_state_idx(pivot_state)
270 if pivot_idx < 0:
271 return []
272 tmp_list.append(pivot_idx)
273
274 # add default children state
275 state = self._state_list[pivot_idx]
276 default_child_name = state.get_default_child_name()
277 while default_child_name != None:
278 idx = self._search_state_idx(default_child_name)
279 tmp_list.append(idx)
280 state = self._state_list[idx]
281 default_child_name = state.get_default_child_name()
282
283 # add parent state
284 state = self._state_list[pivot_idx]
285 parent_name = state.get_parent_name()
286 while (parent_name != None) and (parent_name != ""):
287 idx = self._search_state_idx(parent_name)
288 tmp_list.insert(0, idx)
289 state = self._state_list[idx]
290 parent_name = state.get_parent_name()
